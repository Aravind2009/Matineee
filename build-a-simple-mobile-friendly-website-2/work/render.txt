function renderRoom(r){
 const people=livePeople(r), settled=people.filter(p=>status(r,p)==='Paid').length, remain=people.reduce((a,p)=>a+Math.max(0,owed(r,p.id)-received(r,p.id)),0);
 const main=document.getElementById('main');
 const attendeeRows=people.map(p=>{let paid=received(r,p.id),pending=sent(r,p.id),balance=Math.max(0,owed(r,p.id)-paid),st=status(r,p);return `<div class="person"><div><div class="person-name">${esc(p.name)}</div><div class="person-detail">Owes ${money(owed(r,p.id))} · Paid ${money(paid)}${pending?` · ${money(pending)} marked sent`:''}</div><span class="status ${st==='Paid'?'paid':st==='Unpaid'?'unpaid':'partial'}">${st}</span></div><div class="toolbar"><div class="amount">${money(balance)}<small>remaining</small></div><button class="icon-btn" onclick="editPerson('${p.id}')" aria-label="Edit attendee">✎</button><button class="icon-btn" onclick="dropPerson('${p.id}')" aria-label="Remove attendee">×</button></div></div>`}).join('');
 const bookers=r.bookers.map(b=>`<div class="booker-row"><div><div class="booker-name">${esc(b.name)}</div><div class="booker-detail">${b.tickets} tickets booked</div></div><div class="amount">${money(b.paid)}<small>paid for tickets</small></div></div>`).join('');
 const ledger=r.payments.slice().reverse().map(t=>{let p=r.people.find(x=>x.id===t.person);return `<div class="ledger-item"><div><b>${esc(p?.name||'Removed attendee')}</b> · ${money(Math.abs(t.amount))}${t.amount<0?' · refund':''}<div class="person-detail">${esc(t.method)}${t.note?' · '+esc(t.note):''} · ${t.state==='sent'?'Marked sent':'Confirmed by booker'}</div></div><div class="ledger-actions">${t.state==='sent'?`<button class="button secondary small" onclick="markReceived('${t.id}')">Confirm received</button>`:''}<button class="icon-btn" onclick="editPayment('${t.id}')" aria-label="Edit payment">✎</button><button class="icon-btn" onclick="removePayment('${t.id}')" aria-label="Undo payment">↶</button></div></div>`}).join('');
 const allocations=total(r),booked=r.bookers.reduce((a,b)=>a+b.paid,0);
 main.innerHTML=`<section class="panel room-head"><div><div class="eyebrow">Private movie room · ${esc(r.code)}</div><h2 class="film-title">${esc(r.movie)}</h2><div class="film-meta">${dateText(r.date)}${r.time?' · '+esc(r.time):''} · ${esc(r.cinema||'Cinema to be decided')}</div>${r.notes?`<div class="notes">${esc(r.notes)}</div>`:''}</div><div class="actions"><button class="button ghost small" onclick="shareRoom()">↗ Share</button><button class="button secondary small" onclick="editOuting()">Edit details</button><button class="button primary small" onclick="addPerson()">＋ Person</button></div></section>
 <div class="stats"><div class="panel stat"><div class="stat-label">Total allocated</div><div class="stat-value">${money(allocations)}</div><div class="stat-foot">${people.length} active attendees</div></div><div class="panel stat"><div class="stat-label">Still to receive</div><div class="stat-value">${money(remain)}</div><div class="stat-foot">Confirmed payments only</div></div><div class="panel stat"><div class="stat-label">Settled up</div><div class="stat-value">${settled} / ${people.length}</div></div></div>
 <div class="content-grid"><section class="panel card"><div class="card-head"><div class="card-title">Attendee balances</div><button class="button ghost small" onclick="addPerson()">＋ Add</button></div>${attendeeRows||'<div class="empty">Add attendees to start splitting the cost.</div>'}<div class="hint">Set each person’s allocation explicitly for different ticket prices, discounts, or shared fees. Amounts are kept to the paise. Removed attendees keep their payment history and their allocation is not redistributed automatically.</div></section><section><div class="panel card"><div class="card-head"><div class="card-title">Booked & paid</div><button class="button ghost small" onclick="addBooker()">＋ Booker</button></div>${bookers||'<div class="empty">Add a booker and ticket total.</div>'}<div class="hint">Ticket + fee total <b>${money(allocations)}</b> · Bookers paid <b>${money(booked)}</b>${allocations===booked?'<br>✓ Totals match.':'<br>⚠ Totals differ. Adjust allocations or booker totals to reconcile.'}</div></div><div class="panel card ledger"><div class="card-head"><div class="card-title">Payment activity</div><button class="button primary small" onclick="addPayment()">＋ Record</button></div>${ledger||'<div class="empty">No payments recorded yet.</div>'}</div><div class="panel card" style="margin-top:16px"><div class="card-head"><div class="card-title">Room details</div><button class="button ghost small" onclick="editInstructions()">${r.instructions?'Edit':'Add'}</button></div><div class="person-detail">Instructions are visible to anyone who opens this room on this device.</div>${r.instructions?`<div class="hint"><b>${esc(r.instructions)}</b><br>Only share details you’re comfortable showing to friends.</div>`:'<div class="person-detail" style="margin-top:10px">Never add a PIN, OTP, password, or card details.</div>'}<div class="room-code">Invite code <b>${esc(r.code)}</b><br>Invitation only, not a password. This local demo cannot sync room changes to another phone.</div></div></section></div>
 <div class="notice">Payment tracking only. This app does not send, receive, or verify money. “Sent” is attendee reported; “received” is confirmed by the booker. Data stays in this browser and is not backed up.</div><div class="panel card" style="margin-top:15px"><div class="card-head"><div class="card-title">Who should receive what?</div><button class="button ghost small" onclick="copySummary()">Copy summary</button></div>${r.bookers.map(b=>`<div class="booker-row"><div><div class="booker-name">${esc(b.name)} should receive</div><div class="booker-detail">Share split in proportion to what each booker paid</div></div><div class="amount">${money(collectable(r,b.id))}<small>remaining collection share</small></div></div>`).join('')||'<div class="person-detail">Add a booker to see the collection summary.</div>'}</div>`;
}
function bookerShares(r){let w=r.bookers.map(b=>Math.max(0,b.paid)),sum=w.reduce((a,x)=>a+x,0);return r.bookers.map((b,i)=>({b,weight:sum?w[i]/sum:(r.bookers.length?1/r.bookers.length:0)}))}
function collectable(r,bid){let share=bookerShares(r).find(x=>x.b.id===bid)?.weight||0;return livePeople(r).filter(p=>p.id!==bid).reduce((a,p)=>a+Math.max(0,owed(r,p.id)-received(r,p.id))*share,0)}
